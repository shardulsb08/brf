# MPTCP MP_JOIN harness -- pseudo-syscall descriptions.
#
# Companion design doc: .claude/designs/mptcp_join_harness_design.md
# Executor implementation: executor/common_brf_linux_mptcp.h
#
# These pseudo-syscalls let a fuzz input express a stateful MPTCP
# flow whose cryptographic dependencies (token, HMAC over keys+
# nonces) are computed by the executor from captured wire state.
# Mutation knobs in each syscall let syzkaller mutate specific byte
# ranges *after* the executor has produced the valid form -- so
# mutations are attributable, not blind.
#
# Syzlang grammar quirk: syscall declarations must be on a single
# line (structs may span lines but syscalls may not).  The lines
# below are long; do not wrap them.

include <linux/socket.h>
include <linux/in.h>
include <linux/in6.h>

resource mptcp_pair[intptr]

# syz_mptcp_pair_init: create server + client pair, complete MP_CAPABLE,
# capture cryptographic state.  See design doc Section 5.1.
syz_mptcp_pair_init(server_addr ptr[in, int32], client_addr ptr[in, int32], init_flags flags[mptcp_init_flags], out_state ptr[out, mptcp_pair_state_t]) mptcp_pair

# syz_mptcp_join_subflow: initiate MP_JOIN with valid token + HMAC.
# nonce_mut / hmac_mut: rewrite the just-generated/computed bytes after.
syz_mptcp_join_subflow(pair mptcp_pair, addr_id int8, backup int8[0:1], nonce_mut flags[mptcp_nonce_mut], hmac_mut flags[mptcp_hmac_mut])

# syz_mptcp_drive_traffic: send data over an established subflow,
# optionally rewriting the kernel-emitted DSS option bytes.
syz_mptcp_drive_traffic(pair mptcp_pair, subflow_id int32, data ptr[in, array[int8]], data_len bytesize[data], map_mut flags[mptcp_map_mut])

# syz_mptcp_send_control: inject MP_FAIL / MP_FASTCLOSE / MP_RST.
syz_mptcp_send_control(pair mptcp_pair, subflow_id int32, suboption flags[mptcp_control_subopt])

# syz_mptcp_pair_close: release pool slot, close sockets.
# Explicit close required because mptcp_pair is an intptr (not an fd),
# so syzkaller resource lifecycle cannot free pool slots automatically.
syz_mptcp_pair_close(pair mptcp_pair)

# syz_mptcp_pm_announce: trigger kernel to emit ADD_ADDR option on
# wire.  Exercises path-manager add-addr emit + peer-side parse paths.
syz_mptcp_pm_announce(pair mptcp_pair, addr_id int8, addr int32be, port int16)

# syz_mptcp_pm_remove: trigger kernel to remove an address previously
# tracked by the userspace path manager.  Sibling of pm_announce;
# exercises the inverse path: mptcp_pm_nl_remove_doit ->
# mptcp_pm_remove_addr_entry -> mptcp_remove_anno_list_by_saddr (and
# mptcp_userspace_pm_remove_id_zero_address for addr_id == 0).
# Targets the same anno_list / userspace_pm_local_addr_list cleanup
# machinery whose ADD side produced the kmemleak race fixed in
# findings/001_mptcp_pm_destroy_race/.
syz_mptcp_pm_remove(pair mptcp_pair, addr_id int8)

# syz_mptcp_pm_subflow_destroy: tell the kernel to tear down a
# specific subflow on this msk.  Sibling of join_subflow's
# SUBFLOW_CREATE; exercises mptcp_pm_nl_subflow_destroy_doit and
# the subsequent mptcp_close_ssk / subflow teardown path.  Different
# code area from pm_remove (subflow lifecycle vs addr-list
# lifecycle).  Local/remote ports are passed through host-byte-order
# per the same kernel convention as SUBFLOW_CREATE; the fuzzer's
# port mutations occasionally happen to match a real subflow
# (exercises success path) and mostly don't (exercises tuple-lookup
# failure path).  Both surfaces are interesting.
syz_mptcp_pm_subflow_destroy(pair mptcp_pair, addr_id int8, local_port int16, remote_port int16)

# Out-param struct.  Mirrors `struct brf_mptcp_pair_state_out` in
# executor/common_brf_linux_mptcp.h.  Field order and sizes must
# match exactly.
mptcp_pair_state_t {
	server_fd	int32
	client_fd	int32
	local_key	int64
	remote_key	int64
	token		int32
	csum_enabled	int8
}

# Mutation knob enums.  Values in socket_mptcp_crypto.txt.const;
# constant names must match executor/common_brf_linux_mptcp.h #defines.
mptcp_init_flags = MPTCP_INIT_CSUM_ON, MPTCP_INIT_CSUM_OFF, MPTCP_INIT_DENY_JOIN_ID0
mptcp_nonce_mut = MPTCP_NONCE_NORMAL, MPTCP_NONCE_ZERO, MPTCP_NONCE_FLIP_HIGH, MPTCP_NONCE_FLIP_LOW, MPTCP_NONCE_REPLAY
mptcp_hmac_mut = MPTCP_HMAC_NORMAL, MPTCP_HMAC_ZERO, MPTCP_HMAC_BIT_FLIP, MPTCP_HMAC_TRUNCATE, MPTCP_HMAC_SWAP
mptcp_map_mut = MPTCP_MAP_NORMAL, MPTCP_MAP_STALE_SEQ, MPTCP_MAP_OFF_BY_ONE, MPTCP_MAP_INFINITE, MPTCP_MAP_HOLE
mptcp_control_subopt = MPTCP_CTL_FAIL, MPTCP_CTL_FASTCLOSE, MPTCP_CTL_RST
